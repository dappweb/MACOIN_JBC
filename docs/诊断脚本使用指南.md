## 1. get-user-full-data.cjs

### 用途
查询指定用户的所有链上数据，包括：
- 钱包余额 (MC/JBC)
- 用户基本信息
- 门票信息
- 质押记录
- 直推列表
- 奖励历史

### 使用方法
```bash
node scripts/get-user-full-data.cjs <用户地址>
```

## 2. analyze-reward-discrepancy.cjs

### 用途
分析两个用户之间的奖励差异，用于排查"奖励不对"的问题。

### 使用方法
```bash
node scripts/analyze-reward-discrepancy.cjs
```

### 默认分析地址
脚本内置了两个地址进行对比分析。如需修改，编辑脚本中的：
```javascript
const complainingUser = "0x0F1868dA5C4Fe2Bf3a5fa56fFf51304aE3624d24";
const comparisonUser = "0xD0826C698741dca45b14bAa4f42d352F06415FB5";
```

### 输出内容
- 用户基本信息对比
- 奖励类型统计
- 最近门票购买及奖励分配
- 推荐链分析

## 6. 自定义查询脚本

### 创建临时查询脚本

如需执行特定查询，可创建临时脚本：

```javascript
// temp-query.cjs
const { ethers } = require('ethers');

const RPC_URL = 'https://chain.mcerscan.com/';
const PROTOCOL_ADDRESS = '0x77601aC473dB1195A1A9c82229C9bD008a69987A';

const ABI = [
  // 添加需要的ABI
  'function userInfo(address) view returns (...)',
];

async function query() {
  const provider = new ethers.JsonRpcProvider(RPC_URL);
  const protocol = new ethers.Contract(PROTOCOL_ADDRESS, ABI, provider);
  
  // 执行查询
  const result = await protocol.userInfo('0x...');
  console.log(result);
}

query().catch(console.error);
```

运行：
```bash
node temp-query.cjs
```

## 7. 常用查询代码片段

### 查询用户推荐链

```javascript
async function getReferralChain(protocol, address, maxDepth = 15) {
  let current = address;
  let depth = 0;
  
  while (current !== ethers.ZeroAddress && depth < maxDepth) {
    const info = await protocol.userInfo(current);
    console.log(`L${depth}: ${current} | 团队: ${info.teamCount}`);
    current = info.referrer;
    depth++;
  }
}
```

### 查询所有质押

```javascript
async function getAllStakes(protocol, address) {
  const stakes = [];
  let i = 0;
  
  while (true) {
    try {
      const stake = await protocol.userStakes(address, i);
      stakes.push(stake);
      i++;
    } catch {
      break;
    }
  }
  
  return stakes;
}
```

### 统计奖励类型

```javascript
async function getRewardStats(protocol, address) {
  const events = await protocol.queryFilter(
    protocol.filters.ReferralRewardPaid(address)
  );
  
  const stats = {};
  for (const e of events) {
    const type = e.args.rewardType;
    const mc = e.args.mcAmount;
    stats[type] = (stats[type] || 0n) + mc;
  }
  
  return stats;
}
```

### 查询特定时间范围的事件

```javascript
async function getEventsInRange(protocol, fromBlock, toBlock) {
  const events = await protocol.queryFilter(
    protocol.filters.TicketPurchased(),
    fromBlock,
    toBlock
  );
  
  return events;
}
```

## 8. 调试技巧

### 捕获详细错误

```javascript
try {
  await protocol.someFunction();
} catch (error) {
  console.log('错误消息:', error.message);
  console.log('错误数据:', error.data);
  console.log('错误代码:', error.code);
  
  // 如果是合约错误
  if (error.data) {
    const selector = error.data.slice(0, 10);
    console.log('错误选择器:', selector);
  }
}
```

## 10. 常见问题

### Q: 脚本报错 "Cannot find module 'ethers'"
A: 运行 `npm install` 安装依赖

### Q: 脚本报错 "bad address checksum"
A: 检查地址格式，确保使用正确的校验和格式

### Q: 脚本执行很慢
A: 
1. 可能是网络问题，检查RPC连接
2. 事件查询较多时正常会慢
3. 可以限制查询的区块范围

### Q: 查询结果与前端不一致
A: 
1. 前端可能有缓存，以脚本查询结果为准
2. 确认查询的是同一个地址
3. 检查是否有正在进行的交易
