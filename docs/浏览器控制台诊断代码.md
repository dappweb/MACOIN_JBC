# 浏览器控制台诊断代码

## 使用方法
1. 打开DApp页面
2. 按F12打开开发者工具
3. 切换到"Console"（控制台）标签
4. 复制以下代码并粘贴到控制台
5. 按Enter执行

## 完整诊断代码

```javascript
// ============================================
// JBC添加流动性问题诊断工具
// ============================================

async function diagnoseJBCLiquidityIssue() {
  console.log('🔍 开始诊断JBC添加流动性问题...\n');
  
  try {
    // 1. 获取Web3上下文（需要从React组件中获取）
    // 如果无法直接访问，需要手动设置以下变量：
    // const { protocolContract, jbcContract, account, provider, isOwner } = useWeb3();
    
    // 临时方案：从window对象获取（如果DApp暴露了这些对象）
    const protocolContract = window.protocolContract;
    const jbcContract = window.jbcContract;
    const account = window.account || (await window.ethereum?.request({ method: 'eth_accounts' }))?.[0];
    const provider = window.provider || new ethers.BrowserProvider(window.ethereum);
    
    if (!protocolContract || !jbcContract || !account) {
      console.error('❌ 无法获取合约实例或账户信息');
      console.log('💡 提示：请确保已连接钱包，并且DApp已加载');
      return;
    }
    
    console.log('✅ 合约和账户信息获取成功');
    console.log('   账户地址:', account);
    console.log('   协议合约:', await protocolContract.getAddress());
    console.log('   JBC合约:', await jbcContract.getAddress());
    console.log('');
    
    // 2. 检查权限（Owner）
    console.log('🔐 检查权限...');
    try {
      const owner = await protocolContract.owner();
      const isOwner = owner.toLowerCase() === account.toLowerCase();
      console.log('   合约Owner:', owner);
      console.log('   当前账户:', account);
      if (isOwner) {
        console.log('   ✅ 权限检查通过：当前账户是合约Owner');
      } else {
        console.log('   ❌ 权限检查失败：当前账户不是合约Owner');
        console.log('   💡 解决方案：使用正确的Owner地址');
      }
    } catch (error) {
      console.error('   ❌ 检查Owner失败:', error.message);
    }
    console.log('');
    
    // 3. 检查MC余额
    console.log('💰 检查MC余额...');
    try {
      const mcBalance = await provider.getBalance(account);
      const mcBalanceFormatted = ethers.formatEther(mcBalance);
      console.log('   MC余额:', mcBalanceFormatted, 'MC');
      if (mcBalance === 0n) {
        console.log('   ⚠️  MC余额为0，无法支付Gas费用');
      } else if (mcBalance < ethers.parseEther('0.01')) {
        console.log('   ⚠️  MC余额较低，可能不足以支付Gas费用');
      } else {
        console.log('   ✅ MC余额充足');
      }
    } catch (error) {
      console.error('   ❌ 检查MC余额失败:', error.message);
    }
    console.log('');
    
    // 4. 检查JBC余额
    console.log('🪙 检查JBC余额...');
    try {
      const jbcBalance = await jbcContract.balanceOf(account);
      const jbcBalanceFormatted = ethers.formatEther(jbcBalance);
      console.log('   JBC余额:', jbcBalanceFormatted, 'JBC');
      if (jbcBalance === 0n) {
        console.log('   ⚠️  JBC余额为0，无法添加JBC流动性');
      } else {
        console.log('   ✅ JBC余额充足');
      }
    } catch (error) {
      console.error('   ❌ 检查JBC余额失败:', error.message);
    }
    console.log('');
    
    // 5. 检查JBC授权
    console.log('🔐 检查JBC授权...');
    try {
      const protocolAddress = await protocolContract.getAddress();
      const allowance = await jbcContract.allowance(account, protocolAddress);
      const allowanceFormatted = ethers.formatEther(allowance);
      console.log('   协议合约地址:', protocolAddress);
      console.log('   JBC授权额度:', allowanceFormatted, 'JBC');
      
      // 测试需要200,000 JBC
      const requiredAmount = ethers.parseEther('200000');
      console.log('   需要数量: 200,000 JBC');
      
      if (allowance === 0n) {
        console.log('   ❌ JBC未授权');
        console.log('   💡 解决方案：需要授权JBC代币');
        console.log('   授权代码:');
        console.log(`   await jbcContract.approve("${protocolAddress}", ethers.MaxUint256);`);
      } else if (allowance < requiredAmount) {
        console.log('   ⚠️  JBC授权不足');
        console.log(`   当前授权: ${allowanceFormatted} JBC`);
        console.log(`   需要: 200,000 JBC`);
        console.log('   💡 解决方案：增加授权额度或重新授权');
      } else {
        console.log('   ✅ JBC授权充足');
      }
    } catch (error) {
      console.error('   ❌ 检查JBC授权失败:', error.message);
    }
    console.log('');
    
    // 6. 检查当前池子状态
    console.log('🏊 检查当前池子状态...');
    try {
      const poolMC = await protocolContract.swapReserveMC();
      const poolJBC = await protocolContract.swapReserveJBC();
      console.log('   池子MC储备:', ethers.formatEther(poolMC), 'MC');
      console.log('   池子JBC储备:', ethers.formatEther(poolJBC), 'JBC');
      if (poolMC === 0n && poolJBC === 0n) {
        console.log('   ⚠️  池子为空，需要初始化');
      } else {
        console.log('   ✅ 池子已有流动性');
      }
    } catch (error) {
      console.error('   ❌ 检查池子状态失败:', error.message);
    }
    console.log('');
    
    // 7. 测试静态调用
    console.log('🧪 测试静态调用（模拟添加流动性）...');
    try {
      const testJbcAmount = ethers.parseEther('200000');
      const testMcAmount = 0n;
      
      // 检查权限
      const owner = await protocolContract.owner();
      const isOwner = owner.toLowerCase() === account.toLowerCase();
      
      if (!isOwner) {
        console.log('   ⚠️  跳过测试（当前账户不是Owner）');
      } else {
        try {
          await protocolContract.addLiquidity.staticCall(testJbcAmount, {
            value: testMcAmount
          });
          console.log('   ✅ 静态调用成功，理论上可以添加流动性');
        } catch (staticError) {
          console.error('   ❌ 静态调用失败:', staticError.message);
          if (staticError.reason) {
            console.error('   失败原因:', staticError.reason);
          }
          
          // 解析常见错误
          if (staticError.message.includes('OwnableUnauthorizedAccount')) {
            console.log('   💡 原因：权限不足（不是Owner）');
          } else if (staticError.message.includes('TransferFromFailed') || 
                     staticError.message.includes('allowance')) {
            console.log('   💡 原因：JBC授权不足');
          } else if (staticError.message.includes('InsufficientBalance')) {
            console.log('   💡 原因：余额不足');
          }
        }
      }
    } catch (error) {
      console.error('   ❌ 测试失败:', error.message);
    }
    console.log('');
    
    // 8. 检查网络
    console.log('🌐 检查网络信息...');
    try {
      const network = await provider.getNetwork();
      console.log('   链ID:', network.chainId.toString());
      console.log('   网络名称:', network.name);
      
      // MC Chain 的链ID通常是 88813
      if (network.chainId.toString() === '88813') {
        console.log('   ✅ 网络正确（MC Chain）');
      } else {
        console.log('   ⚠️  网络可能不正确，MC Chain 的链ID应该是 88813');
        console.log('   💡 解决方案：切换到MC Chain网络');
      }
    } catch (error) {
      console.error('   ❌ 检查网络失败:', error.message);
    }
    console.log('');
    
    // 总结
    console.log('📊 诊断总结:');
    console.log('   请根据上述检查结果，确认问题所在。');
    console.log('   常见问题：');
    console.log('   1. 权限问题：当前账户不是合约Owner');
    console.log('   2. 余额问题：MC或JBC余额不足');
    console.log('   3. 授权问题：JBC未授权或授权不足');
    console.log('   4. 网络问题：连接到错误的网络');
    console.log('');
    console.log('💡 如果发现问题，请根据提示的解决方案进行修复。');
    
  } catch (error) {
    console.error('❌ 诊断过程出错:', error);
    console.error('错误详情:', error.message);
  }
}

// 执行诊断
diagnoseJBCLiquidityIssue();
```

## 简化版诊断（如果无法访问合约实例）

如果上面的代码无法运行（因为无法访问React组件），可以使用这个简化版本：

```javascript
// 简化版诊断 - 需要手动设置合约地址
async function simpleDiagnose() {
  const PROTOCOL_ADDRESS = '0x298578A691f10A85f027BDD2D9a8D007540FCBB4'; // 替换为实际地址
  const JBC_ADDRESS = 'YOUR_JBC_CONTRACT_ADDRESS'; // 替换为实际JBC合约地址
  
  const provider = new ethers.BrowserProvider(window.ethereum);
  const signer = await provider.getSigner();
  const account = await signer.getAddress();
  
  // 加载合约ABI（需要从artifacts或ABI文件加载）
  // 这里假设已经有protocolContract和jbcContract实例
  
  console.log('账户:', account);
  console.log('请手动检查：');
  console.log('1. JBC余额');
  console.log('2. JBC授权');
  console.log('3. MC余额');
  console.log('4. 权限（Owner）');
}
```

## 快速检查命令

如果DApp已经暴露了合约实例，可以直接在控制台执行：

```javascript
// 检查JBC授权
const allowance = await jbcContract.allowance(account, await protocolContract.getAddress());
console.log('JBC授权:', ethers.formatEther(allowance));

// 检查JBC余额
const balance = await jbcContract.balanceOf(account);
console.log('JBC余额:', ethers.formatEther(balance));

// 检查MC余额
const mcBalance = await provider.getBalance(account);
console.log('MC余额:', ethers.formatEther(mcBalance));

// 检查权限
const owner = await protocolContract.owner();
console.log('合约Owner:', owner);
console.log('当前账户:', account);
console.log('是否Owner:', owner.toLowerCase() === account.toLowerCase());
```

## 手动授权JBC（如果需要）

```javascript
// 如果授权不足，可以手动授权
const protocolAddress = await protocolContract.getAddress();
const approveTx = await jbcContract.approve(protocolAddress, ethers.MaxUint256);
console.log('授权交易哈希:', approveTx.hash);
await approveTx.wait();
console.log('授权成功！');
```
